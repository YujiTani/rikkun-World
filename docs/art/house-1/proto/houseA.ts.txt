/** 木づちの 待機の 向き（度）。骨 `tool` に いつも かける（x = うしろへ ねかせる、z = 外へ） */
export const ARCH_TOOL_REST = { x: 28, z: -10 } as const;

// ------------------------------------------------------------ うなずき（DESIGN 19.5。1 コマ = 1 ステップ = 1/60 秒）
export const NOD_FRAMES = 20; // 頭。体は 2 コマ おくれて 22 で おわる

function ease(t: number): number {
  return t * t * (3 - 2 * t);
}

/** f コマ目（0〜22）の 角度（度）。head = 頭の 前への かたむき（＋ = 下を 向く）、roll = 頭の 横の かしげ、body = 体の 前への かたむき */
export function nodPose(f: number): { head: number; roll: number; body: number } {
  const keys: Array<[number, number]> = [[0, 0], [4, -9], [8, 26], [12, 26], [20, 0]];
  let head = 0;
  for (let i = 0; i < keys.length - 1; i++) {
    const [f0, v0] = keys[i];
    const [f1, v1] = keys[i + 1];
    if (f >= f0 && f <= f1) {
      const t = (f - f0) / (f1 - f0);
      head = v0 + (v1 - v0) * (i === 1 ? t * t : ease(t)); // 下げる 区間は 加速（こくっ）、ほかは なめらか
    }
  }
  // 体は 2 コマ おくれて 小さく ついていく（MOTION #13: 1 つの 軸・同じ タイミング に しない）
  const g = Math.max(0, f - 2);
  let body = 0;
  if (g > 4 && g <= 8) body = 6 * ease((g - 4) / 4);
  else if (g > 8 && g <= 12) body = 6;
  else if (g > 12 && g <= 20) body = 6 * (1 - ease((g - 12) / 8));
  return { head, roll: head * 0.18, body };
}

/** 骨（three.js の Object3D。前 = −z）に かける。下を 向く = x 軸 まわり に − */
export function applyNod(bones: Record<'head' | 'body' | 'armL' | 'tool', { rotation: { set(x: number, y: number, z: number): unknown } }>, f: number): void {
  const D = Math.PI / 180;
  const p = nodPose(f);
  bones.head.rotation.set(-p.head * D, 0, p.roll * D);
  bones.body.rotation.set(-p.body * D, 0, 0);
  bones.armL.rotation.set(-p.body * 0.8 * D, 0, 0);
  bones.tool.rotation.set(ARCH_TOOL_REST.x * D, 0, ARCH_TOOL_REST.z * D);
}

// ============================================================ 家具 5 つ（形 = 1/9 ブロック、0.5 きざみ。絵 = 16×16 の タイル を 場所で うつす）
// 原点 = 家具の 左前 下の かど（x・z の 小さい 方）。前 = −z（いすに 座る 向き・ほんだなの あいた 面）。
// 絵の うつし方（ローカルの 座標。家具を まわすと 絵も いっしょに まわる。1 ブロック ごとに くり返す）:
//   上・下の 面 (u, v) = (x, −z) ／ −z の 面 (x, y) ／ +z の 面 (−x, y) ／ +x の 面 (z, y) ／ −x の 面 (−z, y)
//   画素 = floor(frac(u)·16)、行 = 15 − floor(frac(v)·16)（絵の 上 = 上）
// 面の 明るさ = ブロックと 同じ（上 1.0・南北 0.68・東西 0.5・下 0.45）。glow の 面は 1.0（あんどんの 紙）。

export type Px = number[]; // 16×16 = 256 の 0xRRGGBB（上の 行 から）

function rng(seed: number): () => number {
  let s = seed >>> 0 || 1;
  return () => ((s = (s * 1664525 + 1013904223) >>> 0) / 4294967296);
}
const H = (s: string): number => parseInt(s.replace('#', ''), 16);
function speck(base: string, others: string[], seed: number, rate = 0.25): Px {
  const r = rng(seed);
  return Array.from({ length: 256 }, () => (r() < rate ? H(others[Math.floor(r() * others.length)]) : H(base)));
}
function rect(t: Px, x: number, y: number, w: number, h: number, c: string): void {
  for (let j = y; j < y + h; j++) for (let i = x; i < x + w; i++) if (i >= 0 && j >= 0 && i < 16 && j < 16) t[j * 16 + i] = H(c);
}
/** 木目（横 か たて の 線 4 本）。ループ の 形は この まま（条件 の 中で r() を よぶ ので 線の 長さが 決まる） */
function grain(base: string, dark: string, light: string, seed: number, vertical = false): Px {
  const t = speck(base, [light], seed, 0.12);
  const r = rng(seed + 7);
  for (let k = 0; k < 4; k++) {
    const p = Math.floor(r() * 16);
    const s = Math.floor(r() * 12);
    for (let q = s; q < Math.min(16, s + 4 + Math.floor(r() * 8)); q++) {
      const [x, y] = vertical ? [p, q] : [q, p];
      t[y * 16 + x] = H(dark);
    }
  }
  return t;
}
function solid(c: string): Px {
  return new Array(256).fill(H(c));
}

}

export interface FurnBox {
  a: V3; // 1/9 ブロック
  b: V3;
  top: HouseTile; // 上（と 下）の 面
  side: HouseTile; // 横の 4 面
  glow?: boolean; // 面の 明るさを かけない
}
export type FurnKind = 'table' | 'chair' | 'andon' | 'shelf' | 'ofuro';
/** 方眼の 大きさ（まわす 前。x × z） */
export const FURN_CELLS: Record<FurnKind, readonly [number, number]> = { table: [2, 1], chair: [1, 1], andon: [1, 1], shelf: [1, 1], ofuro: [2, 1] };

function legs4(x0: number, x1: number, z0: number, z1: number, t: number, h: number): FurnBox[] {
  return [[x0, z0], [x1 - t, z0], [x0, z1 - t], [x1 - t, z1 - t]].map(([x, z]) => ({ a: [x, 0, z] as V3, b: [x + t, h, z + t] as V3, top: 'wood_end', side: 'wood_v' }));
}
