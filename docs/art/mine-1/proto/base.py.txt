# - 9×9×9 の 形（src/world/render/looks/gotsugotsu.ts と common.ts の mix・carved・nugget を Python に 写した）
import math
import numpy as np

T = 16
N = 9
M = N - 1
M32 = 0xFFFFFFFF


# ------------------------------------------------------------------ 色
def hx(c):
    c = c.lstrip('#')
    return (int(c[0:2], 16), int(c[2:4], 16), int(c[4:6], 16))


def lum(c):
    r, g, b = hx(c) if isinstance(c, str) else c
    return 0.2126 * r + 0.7152 * g + 0.0722 * b


# ------------------------------------------------------------------ 16×16 の 絵（textures.ts と 同じ）
def lcg(seed):
    s = [seed & M32]

    def r():
        s[0] = (s[0] * 1664525 + 1013904223) & M32
        return s[0] / 4294967296
    return r


def solid(c):
    return np.tile(np.array(hx(c), np.int32), (T, T, 1))


def speckle(base, dots, seed, count=40):
    t = solid(base)
    r = lcg(seed)
    for i in range(count):
        x = int(r() * T)
        y = int(r() * T)
        t[y, x] = hx(dots[i % len(dots)])
    return t


def calm(t, base, contrast=0.5, saturation=0.8):
    b = np.array(hx(base), np.float64)
    v = b + (t.astype(np.float64) - b) * contrast
    l = 0.2126 * v[..., 0] + 0.7152 * v[..., 1] + 0.0722 * v[..., 2]
    o = l[..., None] + (v - l[..., None]) * saturation
    return np.clip(np.floor(o + 0.5), 0, 255).astype(np.int32)


def blobs(t, color, seed, n):
    t = t.copy()
    r = lcg(seed)
    c = hx(color)
    for i in range(n):
        x = 1 + int(r() * (T - 3))
        y = 1 + int(r() * (T - 3))
        t[y, x] = c
        t[y, x + 1] = c
        t[y + 1, x] = c
        if i % 2 == 0:
            t[y + 1, x + 1] = c
    return t


def stamp(t, x0, y0, rows, cmap):
    """rows の 文字を cmap の 色で おく（' ' や cmap に ない 文字は そのまま）"""
    for j, row in enumerate(rows):
        for i, ch in enumerate(row):
            if ch in cmap:
                x, y = (x0 + i) % T, (y0 + j) % T
                t[y, x] = hx(cmap[ch])
    return t


def rect(t, x0, y0, w, h, c):
    t[y0:y0 + h, x0:x0 + w] = hx(c)
    return t

def cur_stone():

def cur_dirt():
    return calm(speckle('#8B5A2B', ['#6F4520', '#A06A38'], 22), '#8B5A2B')


def cur_sand():
    return calm(speckle('#E8D9A0', ['#D6C487'], 44, 30), '#E8D9A0')


def cur_grass_top():
    return calm(speckle('#5DBB3F', ['#4CA332', '#73CC52'], 11), '#5DBB3F')


def cur_grass_side():
    t = cur_dirt()
    g = cur_grass_top()
    t[0:4] = g[0:4]
    return t


def cur_coal():
    return blobs(cur_stone(), '#222222', 66, 5)


def cur_iron():
    return blobs(cur_stone(), '#E0A872', 77, 5)


def cur_log_side():
    t = solid('#8A5A2E')
    for x in range(T):
        if x % 4 == 1:
            t[:, x] = hx('#6B4421')
    return calm(t, '#8A5A2E')


def cur_log_top():
    t = solid('#8A5A2E')
    for y in range(T):
        for x in range(T):
            d = max(abs(x - 7.5), abs(y - 7.5))
            if d >= 7:
                t[y, x] = hx('#6B4421')
            elif int(d) % 2 == 0:
                t[y, x] = hx('#B88A55')
    return calm(t, '#8A5A2E')


def cur_bedrock():
    return calm(speckle('#3A3A3A', ['#555555'], 88, 50), '#3A3A3A')


def cur_leaves():
    return calm(speckle('#3E9B2F', ['#2E7D22'], 55, 60), '#3E9B2F')


# ------------------------------------------------------------------ 9×9×9 の 形（gotsugotsu.ts）
def mix(a, b):
    h = (((a & M32) ^ 0x9E3779B9) * 0x85EBCA6B) & M32
    h ^= (((b + 0x7F4A7C15) & M32) * 0xC2B2AE35) & M32
    h ^= h >> 15
    h = (h * 0x2C1B3C6D) & M32
    h ^= h >> 12
    return h


def block_hash(salt, bx, by, bz):
    return mix(mix(mix(salt + 1, bx), by), bz)


def dentable(x, y, z, e):
    if y == M and not (e & 1):
        return False
    if y == 0 and not (e & 2):
        return False
    if x == M and not (e & 4):
        return False
    if x == 0 and not (e & 8):
        return False
    if z == M and not (e & 16):
        return False
    if z == 0 and not (e & 32):
        return False
    return True


def carved(x, y, z, e, h, rate, deep, edge):
    if not dentable(x, y, z, e):
        return False
    faces = ((1, M - y, x, z), (2, y, x, z), (4, M - x, z, y), (8, x, z, y), (16, M - z, x, y), (32, z, x, y))
    near = 0
    for f in range(6):
        bit, dd, u, v = faces[f]
        if not (e & bit) or dd > 2:
            continue
        if dd <= 1:
            near += 1
        r = mix(h, f * 16 + (v // 3) * 3 + (u // 3))
        k = r % 9
        if dd == 0 and k < rate:
            return True
        if dd == 1 and k < deep:
            return True
        if dd == 2 and k < deep and u % 3 == 1 and v % 3 == 1 and (r >> 8) % 2 == 0:
            return True
    if near >= 2 and edge > 0:
        r = mix(h, 500 + ((y // 3) * 3 + (z // 3)) * 3 + (x // 3))
        if r % 10 < edge:
            return True
    return False


# 種類ごとの salt（gotsugotsu.ts の kind() を 呼ぶ 順: grass 200, dirt 201, sand 202, stone 203, coal 204, iron 205, log 206, snow 207, bamboo 208, bedrock 209）
SALT = dict(grass=200, dirt=201, sand=202, stone=203, coal=204, iron=205, log=206, bedrock=209)


def grid_from(fn):
    """fn(x, y, z) -> 0/値 で 9×9×9 の 配列（[x, y, z]）"""
    g = np.zeros((N, N, N), np.int16)
    for x in range(N):
        for y in range(N):
            for z in range(N):
                g[x, y, z] = fn(x, y, z)
    return g


def rock_shape(salt, rate, deep, edge):
    def fill(bx, by, bz, e):
        h = block_hash(salt, bx, by, bz)
        return grid_from(lambda x, y, z: 0 if carved(x, y, z, e, h, rate, deep, edge) else 1)
    return fill


def is_nugget(x, y, z, h):
    r = mix(h, ((y // 3) * 3 + (z // 3)) * 3 + (x // 3))
    if r % 5 != 0:
        return False
    return ((x % 3 != 1) + (y % 3 != 1) + (z % 3 != 1)) < 3


def nugget_shape(salt):
    """いまの 鉱石の 形: 石に うまった 3×3×3 の かたまり（まわりの 石だけ けずる）"""
    def fill(bx, by, bz, e):
        h = block_hash(salt, bx, by, bz)
        return grid_from(lambda x, y, z: 1 if is_nugget(x, y, z, h) else (0 if carved(x, y, z, e, h, 5, 2, 0) else 1))
    return fill


def log_shape(bx, by, bz, e):
    h = block_hash(SALT['log'], bx, by, bz)

    def f(x, y, z):
        if y == 0 or y == M or not dentable(x, y, z, e):
            return 1
        off = h % 3
        if (e & 8) and x <= 1 and (z + off) % (2 if x == 0 else 4) == 0:
            return 0
        if (e & 4) and x >= M - 1 and (z + off) % (2 if x == M else 4) == 0:
            return 0
        if (e & 32) and z <= 1 and (x + off) % (2 if z == 0 else 4) == 0:
            return 0
        if (e & 16) and z >= M - 1 and (x + off) % (2 if z == M else 4) == 0:
            return 0
        return 1
    return grid_from(f)


def sand_shape(bx, by, bz, e):
    h = block_hash(SALT['sand'], bx, by, bz)

    def f(x, y, z):
        if (e & 1) and y == M and dentable(x, y, z, e):
            return 0 if (z + ((x // 3) % 2)) % 3 == 0 else 1
        return 0 if carved(x, y, z, e & ~1, h, 3, 1, 3) else 1
    return grid_from(f)


def full_shape(bx, by, bz, e):
    return np.ones((N, N, N), np.int16)


STONE_SHAPE = rock_shape(SALT['stone'], 6, 3, 7)
DIRT_SHAPE = rock_shape(SALT['dirt'], 4, 2, 5)
BEDROCK_SHAPE = rock_shape(SALT['bedrock'], 6, 3, 7)
COAL_SHAPE = nugget_shape(SALT['coal'])
IRON_SHAPE = nugget_shape(SALT['iron'])


# ------------------------------------------------------------------ 面の 座標（面ごとの u, v と 深さ）
# 面の 番号: 0 上(+y), 1 下(-y), 2 東(+x), 3 西(-x), 4 南(+z), 5 北(-z)。bit = 1,2,4,8,16,32
FACE_BITS = (1, 2, 4, 8, 16, 32)


def face_uvd(f, x, y, z):
    """面 f から 見た (u, v, 深さ)。u, v は 面の 上の 位置（0〜8）、深さ 0 = 外がわ。
    u, v の 向きは 絵の 向き（u = 右へ、v = 下へ。横の 面は v = 上から）に そろえる。"""
    if f == 0:
        return x, z, M - y
    if f == 1:
        return x, M - z, y
    if f == 2:
        return M - z, M - y, M - x
    if f == 3:
        return z, M - y, x
    if f == 4:
        return x, M - y, M - z
    return M - x, M - y, z


def face_xyz(f, u, v, d):
    """face_uvd の 逆"""
    if f == 0:
        return u, M - d, v
    if f == 1:
        return u, d, M - v
    if f == 2:
        return M - d, M - v, M - u
    if f == 3:
        return d, M - v, u
    if f == 4:
        return u, M - v, M - d
    return M - u, M - v, d
