def gl_mask(rows, x0, y0, mask, cmap_g):
    for j, row in enumerate(rows):
        for i, ch in enumerate(row):
            if ch in cmap_g:
                mask[(y0 + j) % T, (x0 + i) % T] = cmap_g[ch]
    return mask


def zero_mask():
    return np.zeros((T, T))

# ------------------------------------------------------------------ つらら（十字の 板。仕様 MINE-1 §3.2: 三角形 16 まで・先は まるく 白く）
def icicle_plate(icicles):
    """icicles = [(左の x, [上から 1 行ずつの はば])]。左の はし = 明るい、右の はし = 影、上 = 根もと、
    下の 2 行 = 白い 先（はば 2 いじょう。1 点の とがった 先に しない）"""
    ic = np.zeros((T, T, 4), np.int32)
    for x0, widths in icicles:
        w0 = widths[0]
        n = len(widths)
        for y, w in enumerate(widths):
            s0 = x0 + (w0 - w + 1) // 2
            for i in range(w):
                if y == 0:
                    col = ICE['root']
                elif y >= n - 2:
                    col = ICE_TIP
                elif i == 0 and w >= 3:
                    col = ICE['lt']
                elif i == w - 1 and w >= 2:
                    col = ICE['sh']
                else:
                    col = ICE['main']
                ic[y, s0 + i] = (*hx(col), 255)
    return ic


def icicle_block(prefix, icicles):
    return Block('icicle', Mat(prefix + '_icicle', solid('#FFFFFF')), full_shape, opaque=False, cross=icicle_plate(icicles))


# ------------------------------------------------------------------ 宝箱（3 つの 形。どれも 1 マスより 小さい ＝ たんす・かまどの 四角と 影で ちがう）
def chest_blocks(prefix, style):
    """style: 'cap' = ふたの 大きい 平らな 箱（A）、'urushi' = くろうるし に きん・まるい ふた・金の 足（B）、'round' = かまぼこの ふた（C）。
    あけた = ふたが うしろに 立ち、中は からっぽ（仕様 MINE-1 §3.2）"""
    if style == 'urushi':
        body = Mat(prefix + '_cw', solid(URUSHI['main']))
        lidm = Mat(prefix + '_cl', solid(URUSHI['lid']))
        band = Mat(prefix + '_cb', solid(GOLD['lo']))
        fit = Mat(prefix + '_cf', solid(GOLD['main']))
        inner = Mat(prefix + '_ci', solid(URUSHI['red']))
    elif style == 'cap':
        body = Mat(prefix + '_cw', solid(CHEST['wood']))
        lidm = Mat(prefix + '_cl', solid(CHEST['lid']))
        band = Mat(prefix + '_cb', solid(CHEST['band']))
        fit = Mat(prefix + '_cf', solid(GOLD['main']))
        inner = Mat(prefix + '_ci', solid(CHEST['floor']))
    else:
        body = Mat(prefix + '_cw', calm(speckle(CHEST['wood'], ['#8A5229', '#A8693A'], 171, 30), CHEST['wood']))
        lidm = Mat(prefix + '_cl', calm(speckle(CHEST['lid'], ['#965A30', '#B57444'], 172, 30), CHEST['lid']))
        band = Mat(prefix + '_cb', solid(CHEST['band']))
        fit = Mat(prefix + '_cf', solid(GOLD['main']))
        inner = Mat(prefix + '_ci', solid(CHEST['floor']))
    hole = Mat(prefix + '_ck', solid('#2A1A10'))
    mats = [body, lidm, band, fit, inner, hole]
    W, LD, BD, FT, IN, KH = 1, 2, 3, 4, 5, 6
    y0 = 1 if style == 'urushi' else 0   # B は 金の 足の 上
    yt = y0 + 4                          # 箱の いちばん 上

    def body_box(g, open_):
        g[1:8, y0:yt + 1, 2:7] = W
        for x in (2, 6):
            g[x, y0:yt + 1, 2:7] = BD
        if style == 'urushi':
            for (x, z) in ((1, 2), (7, 2), (1, 6), (7, 6)):
                g[x, 0, z] = FT
                g[x, y0, z] = FT
                g[x, yt, z] = FT
        if open_:
            g[2:7, y0 + 1:yt + 1, 3:6] = 0
            g[2:7, y0, 3:6] = IN
            g[2:7, y0 + 1:yt + 1, 2] = IN   # 中の おくの かべ（前から 見える。からっぽ の 色）

    def lock(g):
        g[4, yt - 1:yt + 1, 7] = FT
        g[4, yt - 1, 7] = KH
        g[4, yt + 1, 7] = FT

    def closed(bx, by, bz, e):
        g = np.zeros((N, N, N), np.int16)
        body_box(g, False)
        if style == 'cap':
            g[0:9, yt + 1:yt + 3, 1:8] = LD
            for x in (2, 6):
                g[x, yt + 1:yt + 3, 1:8] = BD
            g[4, yt - 1:yt + 1, 7] = FT
            g[4, yt - 1, 7] = KH
        else:
            g[1:8, yt + 1:yt + 3, 2:7] = LD
            g[1:8, yt + 3, 3:6] = LD
            for x in (2, 6):
                g[x, yt + 1:yt + 3, 2:7] = BD
                g[x, yt + 3, 3:6] = BD
            lock(g)
        return g

    def opened(bx, by, bz, e):
        g = np.zeros((N, N, N), np.int16)
        body_box(g, True)
        top = min(N - 1, yt + 4)
        if style == 'cap':
            g[0:9, yt + 1:top + 1, 0:2] = LD
            for x in (2, 6):
                g[x, yt + 1:top + 1, 0:2] = BD
        elif style == 'urushi':
            g[1:8, yt + 1:top + 1, 0] = LD
            g[1:8, yt + 1:top + 1, 1] = IN
            for x in (2, 6):
                g[x, yt + 1:top + 1, 0] = BD
        else:
            g[1:8, yt + 1:top + 1, 1] = LD
            g[1:8, yt + 2:top + 1, 0] = LD
            for x in (2, 6):
                g[x, yt + 1:top + 1, 1] = BD
        g[4, yt, 7] = FT
        return g
    return (Block(prefix + '_chest', mats, closed, opaque=False), Block(prefix + '_chest_open', mats, opened, opaque=False))

def tier_chest(prefix, style):
    """段階 2・4 の 宝箱。前 = +z。どれも 1 マスより 小さい（たんす・かまどの 四角と 影で ちがう）。
    あけた = ふたが うしろに 立ち、中は からっぽ（仕様 MINE-1 §3.2）。影の 目じるし: 1 = ふたの ひさし、2 = よこの おび、3 = 金の 足、4 = かんむり"""
    if style == 'iron':
        pal = dict(W=IRONWOOD['body'], L=IRONWOOD['lid'], S=STEEL['main'], D=STEEL['lo'], K=STEEL['dk'], I=IRONWOOD['inside'])
    elif style == 'ironbox':
        pal = dict(W=IRONWOOD['body'], L=IRONWOOD['lid'], S=STEEL['main'], D=STEEL['lo'], K=STEEL['dk'], I=IRONWOOD['inside'])
    elif style == 'steel':
        pal = dict(W=STEEL['lo'], L=STEEL['main'], S=STEEL['dk'], D=STEEL['dk'], K='#1E2026', I='#4A4E58')
    elif style == 'flame':
        pal = dict(W=GOLD['main'], L=GOLD['main'], O=GOLD['lo'], C='#F5CD55', F=MIRACLE['fire'], Y=MIRACLE['core'], T=MIRACLE['tip'], K=GOLD['dk'], I=ROYAL)
    elif style == 'shu':
        pal = dict(W=SHU['main'], L=SHU['lid'], G=GOLD['main'], O=GOLD['lo'], Y=MIRACLE['core'], K=GOLD['dk'], I=URUSHI['main'])
    else:  # wing
        pal = dict(W=GOLD['main'], L=GOLD['main'], O=GOLD['lo'], H=GOLD['hi'], G='#F5CD55', F='#F7F4EC', E='#D9D2C2', K=GOLD['dk'], I=ROYAL,
                   M=MIRACLE['fire'], Y=MIRACLE['core'], T=MIRACLE['tip'])
    keys = list(pal)
    mats = [Mat(f'{prefix}_{k}', solid(c)) for k, c in pal.items()]
    ix = {k: i + 1 for i, k in enumerate(keys)}

    def shape(open_):
        g = np.zeros((N, N, N), np.int16)

        def B(x0, x1, y0, y1, z0, z1, k):  # はしも 入れる
            g[x0:x1 + 1, y0:y1 + 1, z0:z1 + 1] = ix[k] if k else 0

        def P(x, y, z, k):
            g[x, y, z] = ix[k] if k else 0

        def hollow(x0, x1, y0, y1, z0, z1):  # からだ（x0..x1, y0..y1, z0..z1）の 中を あける。床と おくの かべの 内がわ = 中の 色
            B(x0 + 1, x1 - 1, y0 + 1, y1, z0 + 1, z1 - 1, None)
            B(x0 + 1, x1 - 1, y0, y0, z0 + 1, z1 - 1, 'I')
            B(x0 + 1, x1 - 1, y0 + 1, y1, z0, z0, 'I')
        if style in ('iron', 'steel'):
            # はば 9・高さ 7（からだ 4 ＋ かまぼこの ふた 3）。たての おび 2 本が 前・上・うしろを まわって 1 小ブロック とびだす
            B(0, 8, 0, 3, 2, 6, 'W')
            for x in (0, 8):
                for z in (2, 6):
                    P(x, 0, z, 'D')           # 下の かどの 金具
            if open_:
                hollow(0, 8, 0, 3, 2, 6)
                B(0, 8, 4, 8, 1, 1, 'I')      # 立った ふたの うら
                B(0, 8, 4, 8, 0, 0, 'L')
                for x in (2, 6):
                    B(x, x, 0, 3, 7, 7, 'S')
                    B(x, x, 4, 8, 0, 0, 'S')
                    P(x, 1, 7, 'D')
                P(4, 3, 7, 'K')
            else:
                B(0, 8, 4, 5, 2, 6, 'L')      # かまぼこの ふた
                B(0, 8, 6, 6, 3, 5, 'L')
                for x in (2, 6):
                    B(x, x, 0, 5, 7, 7, 'S')
                    P(x, 6, 6, 'S')
                    B(x, x, 6, 6, 3, 5, 'S')
                    B(x, x, 0, 5, 1, 1, 'S')
                    P(x, 6, 2, 'S')
                    P(x, 1, 7, 'D')           # びょう
                    P(x, 4, 7, 'D')
                P(4, 4, 7, 'S')               # じょう前
                P(4, 3, 7, 'K')
        elif style == 'ironbox':
            B(1, 7, 0, 4, 2, 6, 'W')
            for x in (1, 7):
                for z in (2, 6):
                    B(x, x, 0, 4, z, z, 'D')  # かどの 金具（たての ふち）
            for y in (0,) if open_ else (0, 6):
                for x in (0, 8):              # 8 すみの 金具が とびだす
                    for z in (2, 6):
                        P(x, y, z, 'S')
                for x in (1, 7):
                    for z in (1, 7):
                        P(x, y, z, 'S')
            if open_:
                hollow(1, 7, 0, 4, 2, 6)
                B(1, 7, 5, 8, 1, 1, 'I')
                B(1, 7, 5, 8, 0, 0, 'L')
                P(4, 4, 7, 'S')
                P(4, 3, 7, 'K')
            else:
                B(1, 7, 5, 6, 2, 6, 'L')      # 平らな ふた
                for x in (1, 7):
                    for z in (2, 6):
                        B(x, x, 5, 6, z, z, 'D')
                P(4, 5, 7, 'S')               # じょう前
                P(4, 4, 7, 'S')
                P(4, 3, 7, 'K')
        elif style == 'flame':
            # はば 9・おく 7（いちばん 大きい）。からだ 4・ふた 1・上に ほのお 4（5×5 → 3×3 → 3×3 → 先 1）
            B(0, 8, 0, 3, 1, 7, 'W')
            B(0, 8, 0, 0, 1, 7, 'O')          # 下の ふち
            for (x, y, k) in ((3, 1, 'F'), (4, 1, 'F'), (5, 1, 'F'), (3, 2, 'F'), (4, 2, 'Y'), (5, 2, 'F'), (4, 3, 'F')):
                P(x, y, 8, k)                 # 前の ほのおの もよう（とびだす。じょう前 の かわり）
            if open_:
                hollow(0, 8, 0, 3, 1, 7)
                B(0, 8, 4, 8, 1, 1, 'I')      # 立った ふたの うら
                B(0, 8, 4, 8, 0, 0, 'O')
            else:
                B(0, 8, 4, 4, 1, 7, 'O')      # ふた（こい 金の 板）
                B(2, 6, 5, 5, 2, 6, 'F')      # ほのお（宝珠の 形）: 下 5×5 → 3×3 → 3×3 → 先 1。前から 見て しずくの 形
                B(3, 5, 6, 7, 3, 5, 'F')
                P(4, 6, 5, 'Y')               # しん（きいろ）
                P(4, 7, 5, 'Y')
                P(4, 5, 6, 'Y')
                P(4, 8, 4, 'T')               # 先（うすい きいろ）
        elif style == 'shu':
            B(1, 7, 0, 4, 2, 6, 'W')
            for x in (1, 7):
                for z in (2, 6):
                    B(x, x, 0, 4, z, z, 'G')  # 金の かど
            B(1, 7, 0, 0, 2, 6, 'O')
            for (x, y) in ((3, 1), (4, 1), (5, 1), (3, 2), (5, 2), (4, 3)):
                P(x, y, 6, 'G')               # 前の 金の ほのお（ひらたい）
            P(4, 2, 6, 'Y')
            if open_:
                hollow(1, 7, 0, 4, 2, 6)
                B(1, 7, 5, 8, 1, 1, 'I')
                B(1, 7, 5, 8, 0, 0, 'L')
            else:
                B(1, 7, 5, 6, 2, 6, 'L')      # まるい ふた
                B(1, 7, 7, 7, 3, 5, 'L')
                B(1, 7, 5, 5, 2, 6, 'G')      # ふたの 金の ふち
                P(4, 8, 4, 'G')               # 上の つまみ
                P(4, 5, 7, 'G')
                P(4, 4, 7, 'K')
        else:  # wing: きんの 箱 ＋ 左右に しろい はね ＋ にじの 玉
            B(2, 6, 0, 4, 2, 6, 'W')
            B(2, 6, 0, 0, 2, 6, 'O')
            for (xi, xo) in ((1, 0), (7, 8)):    # はね: 外へ 行くほど 上へ（下から 見て V）
                B(xi, xi, 2, 6, 3, 5, 'F')
                B(xo, xo, 4, 8, 3, 5, 'F')
                P(xo, 8, 4, 'G')                  # はねの 先は 金
                P(xi, 2, 4, 'G')
                B(xo, xo, 4, 4, 3, 5, 'E')        # はねの 下の かげ
            for (x, y, k) in ((3, 1, 'M'), (4, 1, 'M'), (5, 1, 'M'), (3, 2, 'M'), (4, 2, 'Y'), (5, 2, 'M'), (4, 3, 'M')):
                P(x, y, 7, k)                     # 前の ほのおの 玉（ミラクルの しるし）
            if open_:
                hollow(2, 6, 0, 4, 2, 6)
                B(2, 6, 5, 8, 1, 1, 'I')
                B(2, 6, 5, 8, 0, 0, 'L')
            else:

# ------------------------------------------------------------------ B・C の つぶの 形（面ごと）
def grain_block(salt, layout, mats, rate=3, deep=1):
    """layout(hf, f) -> (つぶ [(u, v, d, k)], ほる [(u, v, d)])。k = mats の 番号（1 = 地の 石）。
    つぶの まわり 1 マスの 外がわ（d=0）は いつも ほる（みぞ = 影）。ほかの 石は いまの 鉱石と 同じ ほり方。"""
    def fill(bx, by, bz, e):
        h = block_hash(salt, bx, by, bz)
        g = np.ones((N, N, N), np.int16)
        grain = {}
        carve = set()
        for f in range(6):
            if not (e & FACE_BITS[f]):
                continue
            vox, pits = layout(mix(h, 31 + f), f)
            for (u, v, dd, k) in vox:
                if 0 <= u < N and 0 <= v < N:
                    grain[face_xyz(f, u, v, dd)] = k
            occ0 = {(u, v) for (u, v, dd, k) in vox if dd == 0}
            for (u, v) in occ0:
                for du in (-1, 0, 1):
                    for dv in (-1, 0, 1):
                        uu, vv = u + du, v + dv
                        if 0 <= uu < N and 0 <= vv < N and (uu, vv) not in occ0:
                            carve.add(face_xyz(f, uu, vv, 0))
            for (u, v, dd) in pits:
                if 0 <= u < N and 0 <= v < N:
                    carve.add(face_xyz(f, u, v, dd))
        for x in range(N):
            for y in range(N):
                for z in range(N):
                    p = (x, y, z)
                    if p in grain:
                        g[p] = grain[p]
                    elif dentable(x, y, z, e) and (p in carve or carved(x, y, z, e, h, rate, deep, 0)):
                        g[p] = 0
        return g
    return fill


def pick(hf, variants):
    return variants[hf % len(variants)]

def opt_c():
    d = common_blocks()
    st = cur_stone()
    stm = Mat('c_stone_m', st)

    def mats(prefix, cols, glow=None, breath=None):
        out = [stm]
        for k, c in enumerate(cols):
            gv = None if glow is None else np.full((T, T), glow[k])
            out.append(Mat(f'{prefix}{k}', solid(c), glow=gv, breath=breath))
        return out

    def center(hf):
        return 1 + hf % 3, 1 + (hf >> 3) % 3

    def big(shape_rows, hf, kmap):
        """shape_rows: 深さごとの 5×5 の 文字（'.' = なし）"""
        u0, v0 = center(hf)
        vox = []
        for dd, rows in enumerate(shape_rows):
            for j, row in enumerate(rows):
                for i, ch in enumerate(row):
                    if ch in kmap:
                        vox.append((u0 + i, v0 + j, dd, kmap[ch]))

def speckle_on(t, dots, seed, count):
    t = t.copy()
    r = lcg(seed)
    for i in range(count):
        x = int(r() * T)
        y = int(r() * T)
        t[y, x] = hx(dots[i % len(dots)])
    return t
